"""
Day 66 - 模块一：STAR 法则项目经历生成器与简历量化体检 Linter
"""
from __future__ import annotations

import re
from dataclasses import dataclass
from typing import Dict, List

STRONG_VERBS = ("主导", "设计", "架构", "构建", "实现", "重构", "优化", "落地", "搭建")
WEAK_VERBS = ("负责", "参与", "协助", "了解", "学习过", "使用过")
ATS_KEYWORDS = (
    "rag", "bm25", "rerank", "rrf", "agent", "langgraph", "mcp",
    "hitl", "fastapi", "sse", "vllm", "ragas", "docker", "token", "qps", "p95",
)


@dataclass
class StarEntry:
    situation_task: str
    action: str
    result: str

    def render(self) -> str:
        return f"针对{self.situation_task}，{self.action}，最终{self.result}。"


def lint_resume_bullet(bullet: str) -> Dict[str, object]:
    """对单条简历项目经历执行四维体检打分（满分 100）"""
    score = 0
    suggestions: List[str] = []
    low = bullet.lower()

    has_strong = any(v in bullet for v in STRONG_VERBS)
    has_weak = any(v in bullet for v in WEAK_VERBS)
    if has_strong and not has_weak:
        score += 25
    elif has_strong:
        score += 15
        suggestions.append("建议删除'负责/参与'等弱动词，直接用'主导/设计/落地'开头")
    else:
        suggestions.append("缺少强行动动词，建议以'主导设计/构建/优化'等动词突出个人贡献")

    # 检查量化数字（百分比、毫秒、倍数、金额、评分等）
    metrics = re.findall(r"\d+(?:\.\d+)?(?:%|ms|s|倍|元|条|万|k|qps)?", low)
    if len(metrics) >= 2:
        score += 35
    elif len(metrics) == 1:
        score += 20
        suggestions.append("仅有 1 处量化指标，建议补充优化前后对比基线（如从 X% 提升至 Y%）")
    else:
        suggestions.append("缺少量化成果数字！请补充 Recall/Faithfulness/P95延迟/Token降本等指标")

    matched_kw = [kw for kw in ATS_KEYWORDS if kw in low]
    if len(matched_kw) >= 3:
        score += 25
    elif matched_kw:
        score += 12
        suggestions.append("ATS 工程关键词偏少，建议补充具体技术方案词（如 BM25/Rerank/HITL/SSE/Ragas）")
    else:
        suggestions.append("未命中核心 AI 工程关键词，极易被简历初筛过滤")

    if 45 <= len(bullet) <= 160:
        score += 15
    else:
        suggestions.append(f"当前字数 {len(bullet)}，建议控制在 45~160 字以保证高信息密度")

    return {
        "score": score,
        "grade": "EXCELLENT" if score >= 85 else ("GOOD" if score >= 65 else "NEEDS_REWRITE"),
        "matched_keywords": matched_kw,
        "suggestions": suggestions,
    }


if __name__ == "__main__":
    star = StarEntry(
        situation_task="纯向量检索对企业专有错误码召回率偏低(62%)且偶发幻觉的痛点",
        action="主导设计 BM25+向量+RRF 混合检索与 Rerank 置信度拒答流水线，并落地 Ragas 自动化评测门禁",
        result="将 Context Recall 由 62% 提升至 91%，Faithfulness 达 92%，并通过语义缓存与 7B 分级路由降低 54% Token 成本",
    )
    bullet = star.render()
    print("生成 STAR 经历:", bullet)
    print("高分经历体检:", lint_resume_bullet(bullet))
    print("流水账体检:", lint_resume_bullet("负责参与公司知识库项目，使用了大模型和接口开发。"))

"""

from __future__ import annotations

from dataclasses import dataclass
from typing import Dict, List


@dataclass
class ProjectPortfolioSpec:
    project_name: str
    tagline: str
    tech_stack: List[str]
    highlights: List[str]
    benchmarks: Dict[str, str]
    quickstart_cmd: str


def generate_github_readme(spec: ProjectPortfolioSpec) -> str:
    badges = " ".join(f"`{t}`" for t in spec.tech_stack)
    hl_lines = "\n".join(f"- {h}" for h in spec.highlights)
    bm_rows = "\n".join(f"| {k} | {v} |" for k, v in spec.benchmarks.items())
    return (
        f"# {spec.project_name}\n\n"
        f"> {spec.tagline}\n\n"
        f"**核心技术栈**: {badges}\n\n"
        "## 核心工程亮点\n"
        f"{hl_lines}\n\n"
        "## 量化评测与压测指标 (Benchmarks)\n"
        "| 评估维度 | 实测指标 |\n"
        "| --- | --- |\n"
        f"{bm_rows}\n\n"
        "## 一键本地运行与验收\n"
        "```bash\n"
        f"{spec.quickstart_cmd}\n"
        "```\n"
    )

很多零基础转行或者刚毕业的同学进公司后，最容易踩的第一个坑不是代码写不出来，而是“需求没听懂就开始埋头写代码”，写完三天提测才发现跟产品经理要的南辕北辙。今天我们以 STARMIND-066（简历与作品集打磨：STAR 法则项目包装、GitHub 主页 README 与作品集生成器）为例，把星辰数科每周一上午需求评审会的真实对话逻辑拆开揉碎了讲给你听。业务背景非常明确：学员即将投递国内互联网大厂及 AI 独角兽的 AI 应用工程师 / Agent 开发工程师 / LLM 后端岗位，需将 70 天项目成果转化为高转化率简历与可点击验收的 GitHub 作品集。 在评审会上，林产品抛出来的核心业务痛点集中在：（1）简历项目描述多为'负责/参与/使用了某某框架'的流水账，缺少业务痛点与量化指标。；（2）缺少 ATS 岗位核心关键词（如 Hybrid RAG / Rerank / LangGraph / MCP / vLLM / Ragas / SSE / Docker）。；（3）GitHub 仓库没有清晰 README、架构图与性能数据，面试官点进去 3 秒就关掉。。如果你只盯着字面意思去写一个“一次性脚本”，短期看似交付很快，但两周后业务方加一个新字段、换一个大模型厂商或者并发量翻倍时，整段代码就得推倒重来。

老陈在评审会上通常只追问三个致命问题，建议你拿小本本记下来，以后带项目直接套用：第一问“输入数据的最坏情况长什么样？”——比如用户会不会粘进来 5 万字乱码、空文件、或者带 SQL 注入/提示词注入的恶意文本；第二问“业务方能接受的失败兜底策略是什么？”——在大模型应用里，100% 不出错是不可能的，关键是出错时系统是直接 500 白屏，还是友好提示、自动降级并留存日志；第三问“哪些指标能证明我们今天这版上线是成功的？”——围绕今天定下的交付目标：（1）实现 STAR 法则项目经历生成器与简历体检 Linter：自动检测弱动词、缺失量化数字与 ATS 关键词覆盖率。；（2）实现 GitHub 作品集 README 生成器：自动拼装项目亮点、核心技术栈、量化 Benchmark 表格与快速启动命令。，必须全部翻译成可量化的测试用例和监控指标（例如 全量单元测试 100% 通过、核心接口 P95 延迟可控、GitHub 作品集与架构文档可直接用于一线大厂面试）。

经过这三轮对齐，我们在飞书需求文档里落定了今天的技术边界：明确哪些能力放在当天版本 versions/day66/ 里做透做稳，哪些属于过度设计（Over-engineering）暂时留好接口扩展位而不急着实现。记住国内互联网大厂一条铁律：好的架构不是一开始就画一张大而全的宇航级图纸，而是像今天这样，在清晰抽象边界的前提下，让每一个迭代版本都能独立跑通、独立测试、独立交付业务价值。

实战方法论三：从“能跑就行”到“生产级代码”——Day 66 模块的代码品味与重构准则

今天我们在 versions/day66/ 中亲手编写了 src/day66_star_resume_builder.py、src/day66_portfolio_generator.py 以及对应的自动化测试 tests/test_day66.py。站在初级开发者视角，代码只要在控制台打印出预期结果就算大功告成；但站在星辰数科技术委员会的 Code Review 视角，一段代码能不能合入主干，要过四道硬核关卡：第一关是“单一职责与低耦合”——函数长度尽量控制在 40~60 行以内，如果一个函数既负责读取配置、又负责清洗文本、还负责调接口和写文件，那它一定难以复用和单测；第二关是“命名即文档”——变量名绝不用 a、b、data1、temp2，而是直接用业务含义命名（如 sanitized_prompt、chunk_overlap、trace_id），配合 Python 类型注解（Type Hints），让阅读者一眼看清数据契约；第三关是“显式错误处理”——绝不写光秃秃的 except: pass 吞掉异常，因为吞掉异常等于在生产环境埋下一颗定时炸弹，真出故障时连日志都查不到；第四关是“可测试性”——任何涉及外部网络、系统时间、随机数或付费大模型 API 的逻辑，都要支持通过依赖注入或 local_mock 切换在离线环境 100% 确定性地跑通回归测试。

对照这四道关卡回头审视你今天写下的每一行代码，你会清晰感受到：大模型应用开发绝不是“调个 API 写两句提示词”那么肤浅的玩具活，它本质上是“扎实的软件工程底座 + 大模型概率特性治理 + 垂直业务场景洞察”的三位一体。国内企业现在最缺、薪资给得最高的，恰恰是既懂 Python/FastAPI/Docker 工程落地，又懂 RAG/Agent/微调底层调优的复合型 AI 应用工程师。

9. 生产环境排障实录与工程验收自查表

故障演练一：简历与作品集打磨：STAR 法则项目包装、GitHub 主页 README 与作品集生成器 核心链路在异常脏数据或边界输入下触发未预期中断

【线上现象】在联调 STARMIND-066 对应的功能模块时，开发机上用自己构造的“干净样例”跑得丝滑无比，但一交给阿杰跑边界测试（如空文本、超长中文混排、包含特殊 Emoji 或嵌套缺失字段的 JSON），控制台直接抛出未捕获异常，导致整个服务进程或上游调用链雪崩。

【根因定位】根本原因在于把“理想假设”当成了“生产契约”。在 全栈架构整合、全链路自动化测试与压测、生产部署上线、技术答辩与 STAR 简历打磨、高频八股与日活十万级系统设计 环节，上游业务系统传过来的数据往往充满历史包袱（比如字段值为 None 而非空字符串、编码混杂 GBK/UTF-8、或者模型偶发少输出一个右括号）。如果函数入口缺少显式的参数防御与 Pydantic/类型校验，异常就会在深层调用栈里爆炸。

【止血与根治方案】

1. 在所有对外暴露的入口函数与接口层建立“防御性边界”，先校验、清洗并标准化输入，再进入核心逻辑；
2. 为当日核心模块补充至少 3 组异常/空值/极值单元测试，并纳入 tests/test_day66.py 回归集合；
3. 捕获可预期异常时，返回结构化错误对象并附带 trace_id，方便在飞书告警群里一眼定位到具体是哪条请求出了岔子。

故障演练二：性能退化、成本突增与线上可观测性盲区（模块联调接口契约不一致、上线前缺乏端到端回归测试、面试中对项目指标和踩坑细节讲不清楚）

【线上现象】上线试运行两天后，林产品反馈早高峰时段页面转圈变慢，同时财务侧看到大模型 API 账单曲线出现异常毛刺，但翻开服务器日志却只看到零星几句 `print('ok')`，完全无法还原当时到底哪个环节耗时最长、哪条 Prompt 吃掉了最多 Token。

【根因定位】缺少工程化的日志分级、耗时打点与配额熔断机制。很多从零基础转行过来的同学习惯用 `print` 调试完就留在线上，既没有时间戳和请求追踪号，也没有统计输入/输出 Token 数和单步耗时；一旦外部接口抖动或触发死循环重试，成本就会在几分钟内失控。

【止血与根治方案】

1. 严格使用标准 logging 模块替代裸 print，记录关键节点的入参摘要、耗时毫秒数（latency_ms）和状态码；
2. 针对所有外部 I/O（文件、数据库、HTTP/LLM API）显式设置 timeout 超时上限和最大重试次数（通常 <= 3 次 + 指数退避）；
3. 定期对照今日 SLA 基线（全量单元测试 100% 通过、核心接口 P95 延迟可控、GitHub 作品集与架构文档可直接用于一线大厂面试）跑压测回归，把隐性瓶颈消灭在灰度发布之前。

故障演练三：国内团队跨环境一致性问题（开发环境能跑，测试/Docker 环境报错）

【线上现象】学员在自己 Windows/Mac 笔记本的 VS Code 里运行 Day 66 代码一切正常，但提交到 GitLab CI（Linux Debian 容器环境）后，出现文件路径分隔符报错、时区偏差或国内网络下载依赖超时。

【根因定位】代码中硬编码了反斜杠路径（如 `data\test.json`）或绝对路径（如 `C:\Users\...`），未统一使用 `pathlib.Path`；同时 `pip install` 未指定国内镜像源或 `requirements.txt` 未锁定主版本号。

【止血与根治方案】

1. 全仓库统一使用 pathlib.Path(__file__).resolve().parent 构建相对路径，彻底告别跨平台路径坑；
2. 打开文本文件强制显式传参 encoding='utf-8'，避免 Windows 默认 GBK 编码引发的 UnicodeDecodeError；
3. 依赖安装统一配置清华源 -i https://pypi.tuna.tsinghua.edu.cn/simple，并在 README.md 中固定 Python 3.11+ 版本要求。

def calc_jd_match(resume_text: str, jd_keywords: list[str]) -> dict:
    low = resume_text.lower()
    hit = [k for k in jd_keywords if k.lower() in low]
    miss = [k for k in jd_keywords if k.lower() not in low]
    return {"match_rate": round(len(hit) / max(1, len(jd_keywords)), 4), "missing": miss}

def group_portfolio_projects(projects: list[dict]) -> dict[str, list[str]]:
    grouped: dict[str, list[str]] = {}
    for p in projects:
        grouped.setdefault(p["category"], []).append(p["name"])
    return grouped
