"""
Day 65 - 模块一：15 分钟毕业答辩结构校验器与企业评委评分模型
"""
from __future__ import annotations

from dataclasses import dataclass
from typing import Dict, List


@dataclass
class DefenseSection:
    name: str
    minutes: int
    is_core_tech: bool


@dataclass
class JudgeScoreCard:
    architecture: float   # 0~100 (权重 30%)
    live_demo: float      # 0~100 (权重 25%)
    metrics_cost: float   # 0~100 (权重 25%)
    qa_defense: float     # 0~100 (权重 20%)

    def weighted_total(self) -> float:
        total = (
            self.architecture * 0.30
            + self.live_demo * 0.25
            + self.metrics_cost * 0.25
            + self.qa_defense * 0.20
        )
        return round(total, 2)

    def grade(self) -> str:
        s = self.weighted_total()
        if s >= 90:
            return "优秀 (S)"
        if s >= 80:
            return "良好 (A)"
        if s >= 70:
            return "合格 (B)"
        return "需复辩 (C)"


def validate_defense_plan(sections: List[DefenseSection], target_minutes: int = 15) -> Dict[str, object]:
    total = sum(s.minutes for s in sections)
    core_minutes = sum(s.minutes for s in sections if s.is_core_tech)
    core_ratio = round(core_minutes / max(1, total), 4)
    issues: List[str] = []
    if total != target_minutes:
        issues.append(f"总时长应为 {target_minutes} 分钟，当前为 {total} 分钟")
    if core_ratio < 0.65:
        issues.append(f"核心技术与演示占比 {core_ratio:.0%} 低于 65% 红线，背景铺垫过长")
    return {"valid": len(issues) == 0, "total_minutes": total, "core_ratio": core_ratio, "issues": issues}

"""
Day 65 - 模块二：防翻车现场 Demo 编排器（实时主链路 + 确定性降级快照）
"""
from __future__ import annotations

from dataclasses import dataclass
from typing import Callable, List


@dataclass
class DemoStep:
    scene_name: str
    live_fn: Callable[[], str]
    fallback_snapshot: str
    expected_keyword: str


@dataclass
class DemoStepOutcome:
    scene_name: str
    mode: str  # LIVE | FALLBACK
    output: str
    passed: bool


def run_demo_rehearsal(steps: List[DemoStep]) -> List[DemoStepOutcome]:
    outcomes: List[DemoStepOutcome] = []
    for st in steps:
        try:
            out = st.live_fn()
            mode = "LIVE"
        except Exception:
            out = st.fallback_snapshot
            mode = "FALLBACK"
        passed = st.expected_keyword in out
        outcomes.append(DemoStepOutcome(st.scene_name, mode, out, passed))
    return outcomes

实战方法论三：从“能跑就行”到“生产级代码”——Day 65 模块的代码品味与重构准则

今天我们在 versions/day65/ 中亲手编写了 src/day65_defense_planner.py、src/day65_live_demo_runner.py 以及对应的自动化测试 tests/test_day65.py。站在初级开发者视角，代码只要在控制台打印出预期结果就算大功告成；但站在星辰数科技术委员会的 Code Review 视角，一段代码能不能合入主干，要过四道硬核关卡：第一关是“单一职责与低耦合”——函数长度尽量控制在 40~60 行以内，如果一个函数既负责读取配置、又负责清洗文本、还负责调接口和写文件，那它一定难以复用和单测；第二关是“命名即文档”——变量名绝不用 a、b、data1、temp2，而是直接用业务含义命名（如 sanitized_prompt、chunk_overlap、trace_id），配合 Python 类型注解（Type Hints），让阅读者一眼看清数据契约；第三关是“显式错误处理”——绝不写光秃秃的 except: pass 吞掉异常，因为吞掉异常等于在生产环境埋下一颗定时炸弹，真出故障时连日志都查不到；第四关是“可测试性”——任何涉及外部网络、系统时间、随机数或付费大模型 API 的逻辑，都要支持通过依赖注入或 local_mock 切换在离线环境 100% 确定性地跑通回归测试。

9. 生产环境排障实录与工程验收自查表

故障演练一：毕业答辩日：15 分钟高分答辩结构、企业评委追问清单与防翻车演示验收脚本 核心链路在异常脏数据或边界输入下触发未预期中断

【线上现象】在联调 STARMIND-065 对应的功能模块时，开发机上用自己构造的“干净样例”跑得丝滑无比，但一交给阿杰跑边界测试（如空文本、超长中文混排、包含特殊 Emoji 或嵌套缺失字段的 JSON），控制台直接抛出未捕获异常，导致整个服务进程或上游调用链雪崩。

【根因定位】根本原因在于把“理想假设”当成了“生产契约”。在 全栈架构整合、全链路自动化测试与压测、生产部署上线、技术答辩与 STAR 简历打磨、高频八股与日活十万级系统设计 环节，上游业务系统传过来的数据往往充满历史包袱（比如字段值为 None 而非空字符串、编码混杂 GBK/UTF-8、或者模型偶发少输出一个右括号）。如果函数入口缺少显式的参数防御与 Pydantic/类型校验，异常就会在深层调用栈里爆炸。

【止血与根治方案】

1. 在所有对外暴露的入口函数与接口层建立“防御性边界”，先校验、清洗并标准化输入，再进入核心逻辑；
2. 为当日核心模块补充至少 3 组异常/空值/极值单元测试，并纳入 tests/test_day65.py 回归集合；
3. 捕获可预期异常时，返回结构化错误对象并附带 trace_id，方便在飞书告警群里一眼定位到具体是哪条请求出了岔子。

故障演练二：性能退化、成本突增与线上可观测性盲区（模块联调接口契约不一致、上线前缺乏端到端回归测试、面试中对项目指标和踩坑细节讲不清楚）

【线上现象】上线试运行两天后，林产品反馈早高峰时段页面转圈变慢，同时财务侧看到大模型 API 账单曲线出现异常毛刺，但翻开服务器日志却只看到零星几句 `print('ok')`，完全无法还原当时到底哪个环节耗时最长、哪条 Prompt 吃掉了最多 Token。

【根因定位】缺少工程化的日志分级、耗时打点与配额熔断机制。很多从零基础转行过来的同学习惯用 `print` 调试完就留在线上，既没有时间戳和请求追踪号，也没有统计输入/输出 Token 数和单步耗时；一旦外部接口抖动或触发死循环重试，成本就会在几分钟内失控。

【止血与根治方案】

1. 严格使用标准 logging 模块替代裸 print，记录关键节点的入参摘要、耗时毫秒数（latency_ms）和状态码；
2. 针对所有外部 I/O（文件、数据库、HTTP/LLM API）显式设置 timeout 超时上限和最大重试次数（通常 <= 3 次 + 指数退避）；
3. 定期对照今日 SLA 基线（全量单元测试 100% 通过、核心接口 P95 延迟可控、GitHub 作品集与架构文档可直接用于一线大厂面试）跑压测回归，把隐性瓶颈消灭在灰度发布之前。

故障演练三：国内团队跨环境一致性问题（开发环境能跑，测试/Docker 环境报错）

【线上现象】学员在自己 Windows/Mac 笔记本的 VS Code 里运行 Day 65 代码一切正常，但提交到 GitLab CI（Linux Debian 容器环境）后，出现文件路径分隔符报错、时区偏差或国内网络下载依赖超时。

from src.day65_live_demo_runner import DemoStepOutcome


def calc_live_stability(outcomes: list[DemoStepOutcome]) -> float:
    ok = sum(1 for o in outcomes if o.mode == "LIVE" and o.passed)
    return round(ok / max(1, len(outcomes)), 4)

    DemoStepOutcome("s1", "LIVE", "ok", True),
    DemoStepOutcome("s2", "LIVE", "ok", True),

])
